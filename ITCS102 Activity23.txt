def my_factorial(n):
    fact = 1
    for i in range(n, 0, -1):
        fact *= i
    return fact


def greet_me(name):
    greet = "hi" + name
    print(greet)


def box(num):
    for x in range(num+1):
        for y in range(num+1):
            if x == 0 or x == num or y == 0 or y == num:
                print(y, end=" ")
            else:
                print(" ", end=" ")
        print()
